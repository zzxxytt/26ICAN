cmake_minimum_required(VERSION 3.8)
project(recognize_yolo)

# ===========================================================================
# 临时诊断版 CMakeLists.txt
#
# 目的：确认这份 CMakeLists 在一次 configure 里被执行了几遍。
# 用全局属性（不是普通变量）计数，全局属性不受 CMake 目录作用域影响，
# 所以嵌套 add_subdirectory 或变量作用域问题都能被如实反映。
#
# 预期与结论：
#   CONFIGURE_COUNT=1 且构建成功 -> 原来的报错来自别处（多为构建缓存）
#   CONFIGURE_COUNT=2            -> CMakeLists 被执行了两遍，双重配置问题
# ===========================================================================
get_property(_configure_count GLOBAL PROPERTY RECOGNIZE_YOLO_CONFIGURE_COUNT)
if(NOT _configure_count)
  set(_configure_count 0)
endif()
math(EXPR _configure_count "${_configure_count} + 1")
set_property(GLOBAL PROPERTY RECOGNIZE_YOLO_CONFIGURE_COUNT ${_configure_count})

# 同时把关键路径与变量打进 CMakeCache，便于事后核对
set(RY_DIAG_COUNT ${_configure_count} CACHE STRING "recognize_yolo: configure 执行次数" FORCE)

message(STATUS "==================== recognize_yolo 诊断 ====================")
message(STATUS "CONFIGURE_COUNT (全局计数) = ${_configure_count}")
message(STATUS "CMAKE_CURRENT_LIST_DIR     = ${CMAKE_CURRENT_LIST_DIR}")
message(STATUS "CMAKE_CURRENT_SOURCE_DIR   = ${CMAKE_CURRENT_SOURCE_DIR}")
message(STATUS "PROJECT_NAME               = ${PROJECT_NAME}")
message(STATUS "CMAKE_CURRENT_BINARY_DIR   = ${CMAKE_CURRENT_BINARY_DIR}")
message(STATUS "AMENT_CMAKE_SYMLINK_INSTALL= ${AMENT_CMAKE_SYMLINK_INSTALL}")
message(STATUS "BUILD_TESTING              = ${BUILD_TESTING}")
message(STATUS "===========================================================")

# 检查根目录是否存在会让 CMake 自动 add_subdirectory 的残留标志
foreach(_stray setup.py setup.cfg pyproject.toml)
  if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/${_stray}")
    message(WARNING "认到根目录残留文件 ${_stray} —— 它可能让 CMake 把子目录当独立项目再配置一遍")
  endif()
endforeach()
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/${PROJECT_NAME}/CMakeLists.txt")
  message(WARNING "认到 ${PROJECT_NAME}/CMakeLists.txt —— 子包被当成子项目了！这就是双重配置的直接原因")
endif()

option(RECOGNIZE_YOLO_NPU "Build the VIPLite NPU worker" ON)
option(RECOGNIZE_YOLO_NPU_ROOT "Root of the VIPLite runtime copied from the AW model zoo" "")

set(NPU_WORKER_TARGET "")

if(RECOGNIZE_YOLO_NPU)
  find_package(OpenCV QUIET COMPONENTS core imgproc imgcodecs)
  find_package(Threads REQUIRED)

  find_path(VIP_LITE_INCLUDE_DIR
    NAMES vip_lite.h
    HINTS
      ${RECOGNIZE_YOLO_NPU_ROOT}/include
      /usr/include
      /usr/local/include
      /opt/vip/include
  )

  find_library(VIP_LITE_LIBRARY
    NAMES vip-lite vip_lite
    HINTS
      ${RECOGNIZE_YOLO_NPU_ROOT}/lib
      ${RECOGNIZE_YOLO_NPU_ROOT}/lib_linux_aarch64/A733
      /usr/lib
      /usr/local/lib
      /usr/lib/aarch64-linux-gnu
  )

  if(VIP_LITE_INCLUDE_DIR AND VIP_LITE_LIBRARY)
    message(STATUS "recognize_yolo: VIPLite 已找到, 将编译 NPU worker")
    add_executable(yolo_npu_worker
      src/yolo_npu_worker.cpp
      src/vip_runner.cpp
      src/postprocess.cpp
    )
    target_include_directories(yolo_npu_worker PRIVATE
      ${CMAKE_CURRENT_SOURCE_DIR}/src
      ${VIP_LITE_INCLUDE_DIR}
    )
    target_link_libraries(yolo_npu_worker PRIVATE
      ${VIP_LITE_LIBRARY}
      ${OpenCV_LIBS}
      Threads::Threads
    )
    set(NPU_WORKER_TARGET yolo_npu_worker)
  else()
    message(WARNING
      "recognize_yolo: 未找到 VIPLite 运行时 (vip_lite.h / libvip-lite.so)。\n"
      "  跳过 yolo_npu_worker 的编译。")
  endif()
endif()
